<?php

namespace App\Controllers\Admin;

use App\Controllers\BaseController;
use App\Models\CompanyModel;
use App\Models\CategoryModel;
use App\Models\LocationModel;
use App\Models\EnquiryModel;

class Dashboard extends BaseController
{
    public function index()
    {
        $companyModel = new CompanyModel();
        $enquiryModel = new EnquiryModel();
        
        $data = [
            'title' => 'Admin Dashboard - travel360',
            'total_companies' => $companyModel->countAll(),
            'pending_listings' => $companyModel->where('status', 'pending')->findAll(5),
            'recent_enquiries' => $enquiryModel->orderBy('created_at', 'DESC')->findAll(5),
        ];
        
        return view('admin/dashboard', $data);
    }
}

class Companies extends BaseController
{
    public function index()
    {
        $model = new CompanyModel();
        $data = [
            'title' => 'Manage Companies - travel360',
            'companies' => $model->findAll(),
        ];
        return view('admin/companies', $data);
    }

    public function approve($id)
    {
        $model = new CompanyModel();
        $model->update($id, ['status' => 'approved']);
        return redirect()->back()->with('success', 'Company approved successfully.');
    }
}

class Categories extends BaseController
{
    public function index()
    {
        $model = new CategoryModel();
        $data = [
            'title' => 'Manage Categories - travel360',
            'categories' => $model->findAll(),
        ];
        return view('admin/categories', $data);
    }
}
